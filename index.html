Chatbox

Chat with anyone using just a username. No email or phone number.

Demo mode: setup for real use

Right now accounts and chats stay inside this browser, so you can try it by opening two tabs and making two accounts. To chat across phones and laptops:

  1. Create a free project at console.firebase.google.com.
  2. Build → Authentication → turn on Email/Password (users never see or type an email).
  3. Build → Firestore Database → create, then paste these rules and publish.
  4. Project settings → add a Web app, copy its config into FIREBASE_CONFIG at the top of this file's script, then upload the file (for example to GitHub Pages).
rules_version = '2';
service cloud.firestore {
 match /databases/{d}/documents {
  match /usernames/{u} {
   allow read: if request.auth != null;
   allow create: if request.auth != null
    && request.auth.token.email == u + '@chatbox.local'
    && request.resource.data.uid == request.auth.uid;
  }
  match /chats/{c} {
   allow read: if request.auth != null && request.auth.uid in resource.data.members;
   allow create: if request.auth != null
    && request.auth.uid in request.resource.data.members
    && request.resource.data.requester == request.auth.uid
    && request.resource.data.status == 'pending';
   allow update: if request.auth != null
    && request.auth.uid in resource.data.members
    && request.resource.data.members == resource.data.members
    && (request.resource.data.status == resource.data.status
        || request.auth.uid != resource.data.requester);
   match /messages/{m} {
    allow read: if request.auth != null
     && request.auth.uid in get(/databases/$(d)/documents/chats/$(c)).data.members;
    allow create: if request.auth != null
     && request.resource.data.from == request.auth.uid
     && request.auth.uid in get(/databases/$(d)/documents/chats/$(c)).data.members
     && get(/databases/$(d)/documents/chats/$(c)).data.status == 'accepted';
   }
  }
 }
}
